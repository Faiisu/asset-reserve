// =====================================================
// USERS
// =====================================================

Table users {
  id uuid [pk]

  email varchar(255) [not null, unique]
  password_hash text [note: 'NULL ได้สำหรับ user ที่ใช้ external authentication only']

  is_active boolean [not null, default: true]

  created_at timestamptz [not null, default: `now()`]
  updated_at timestamptz [not null, default: `now()`]
}


Table user_profiles {
  user_id uuid [pk]

  name varchar(150) [not null]
  phone_number varchar(30)
  profile_picture_url text

  created_at timestamptz [not null, default: `now()`]
  updated_at timestamptz [not null, default: `now()`]
}


Table user_auth_identities {
  id uuid [pk]

  user_id uuid [not null]

  provider varchar(30) [not null, note: 'google, microsoft, etc.']
  provider_user_id varchar(255) [not null]

  created_at timestamptz [not null, default: `now()`]

  indexes {
    (provider, provider_user_id) [unique]
    (user_id, provider) [unique]
  }
}


// =====================================================
// MANAGEMENT AREA
// =====================================================

Table management_areas {
  id uuid [pk]

  name varchar(150) [not null, unique]
  description text

  created_at timestamptz [not null, default: `now()`]
  updated_at timestamptz [not null, default: `now()`]
}


// =====================================================
// RBAC
// =====================================================

Table roles {
  id uuid [pk]

  name varchar(50) [not null, unique, note: '''
    ADMIN
    MANAGER
    USER
  ''']

  description text

  permissions jsonb [not null, default: `[]`, note: '''
    ตัวอย่าง:

    ADMIN:
    [
      "asset:create",
      "asset:view",
      "asset:update",
      "asset:delete",

      "reservation:create",
      "reservation:update",
      "reservation:approve",
      "reservation:reject",

      "return:approve",

      "user:create",
      "user:disable",

      "role:assign"
    ]

    MANAGER:
    [
      "asset:create",
      "asset:view",
      "asset:update",

      "reservation:create",
      "reservation:update",
      "reservation:approve",
      "reservation:reject",

      "return:approve"
    ]

    USER:
    [
      "asset:view",
      "reservation:create",
      "reservation:update"
    ]
  ''']

  created_at timestamptz [not null, default: `now()`]
  updated_at timestamptz [not null, default: `now()`]
}


Table user_roles {
  id uuid [pk]

  user_id uuid [not null]
  role_id uuid [not null]

  management_area_id uuid [note: '''
    NULL = Global Role

    ตัวอย่าง:
    ADMIN + NULL
    = Admin ทั้งระบบ

    MANAGER + Area A
    = Manager เฉพาะ Area A
  ''']

  assigned_by uuid

  created_at timestamptz [not null, default: `now()`]

  indexes {
    (user_id, role_id, management_area_id) [unique]
  }
}


// IMPORTANT:
//
// PostgreSQL UNIQUE ปกติยอมให้ NULL ซ้ำได้
//
// migration จริงควรมี:
//
// UNIQUE (user_id, role_id, management_area_id)
// WHERE management_area_id IS NOT NULL
//
// และ
//
// UNIQUE (user_id, role_id)
// WHERE management_area_id IS NULL
//
// หรือ PostgreSQL 15+ ใช้ NULLS NOT DISTINCT


// =====================================================
// ASSETS
// =====================================================

Table assets {
  id uuid [pk]

  management_area_id uuid [not null]

  name varchar(150) [not null]
  description text
  picture_url text

  total_quantity int [not null, default: 0, note: 'Current physical quantity; ต้อง >= 0']

  is_active boolean [not null, default: true]

  created_at timestamptz [not null, default: `now()`]
  updated_at timestamptz [not null, default: `now()`]

  indexes {
    (management_area_id, name) [unique]
  }
}


// =====================================================
// ASSET TRANSACTION BATCH
// =====================================================

Table asset_transaction_batches {
  id uuid [pk]

  description text

  performed_by uuid [not null]

  created_at timestamptz [not null, default: `now()`]
}


// =====================================================
// INVENTORY LEDGER
// =====================================================

Table asset_transactions {
  id uuid [pk]

  batch_id uuid [not null]
  asset_id uuid [not null]

  transaction_type varchar(30) [not null, note: '''
    ADD
    REMOVE
    LOST
    BROKEN
    REPAIR
    RETURN
    ADJUSTMENT
  ''']

  quantity_delta int [not null, note: '''
    Signed quantity.

    ADD        = positive
    REMOVE     = negative
    LOST       = negative
    BROKEN     = negative
    RETURN     = positive

    REPAIR / ADJUSTMENT
    ขึ้นอยู่กับการเปลี่ยน quantity จริง
  ''']

  reason text

  created_at timestamptz [not null, default: `now()`]

  indexes {
    (asset_id, created_at)
    (batch_id)
  }
}


// =====================================================
// RESERVATION
// =====================================================

Table reservations {
  id uuid [pk]

  requester_id uuid [not null]

  event_name varchar(200) [not null]
  description text

  start_at timestamptz [not null]
  end_at timestamptz [not null]

  status varchar(30) [not null, default: 'DRAFT', note: '''
    DRAFT
    PENDING_APPROVAL
    PARTIALLY_APPROVED
    APPROVED
    REJECTED
    CANCELLED
    ACTIVE
    COMPLETED
  ''']

  created_at timestamptz [not null, default: `now()`]
  updated_at timestamptz [not null, default: `now()`]

  indexes {
    (requester_id, created_at)
    (start_at, end_at)
    (status)
  }
}


// CHECK:
// end_at > start_at


// =====================================================
// RESERVATION ITEMS
// =====================================================

Table reservation_items {
  id uuid [pk]

  reservation_id uuid [not null]
  asset_id uuid [not null]

  quantity int [not null, note: 'ต้อง > 0']

  status varchar(30) [not null, default: 'PENDING', note: '''
    PENDING
    APPROVED
    REJECTED
    PREPARED
    CHECKED_OUT
    RETURNED
  ''']

  prepared_by uuid
  prepared_at timestamptz

  checked_out_by uuid
  checked_out_at timestamptz

  created_at timestamptz [not null, default: `now()`]
  updated_at timestamptz [not null, default: `now()`]

  indexes {
    (reservation_id, asset_id) [unique]

    (asset_id, status)
    (reservation_id, status)
  }
}


// =====================================================
// RESERVATION PARTICIPANTS
// =====================================================

Table reservation_participants {
  reservation_id uuid [not null]
  user_id uuid [not null]

  created_at timestamptz [not null, default: `now()`]

  indexes {
    (reservation_id, user_id) [pk]
  }
}


// =====================================================
// RESERVATION ITEM APPROVAL
// =====================================================

Table reservation_item_approvals {
  id uuid [pk]

  reservation_item_id uuid [not null, unique]

  approver_id uuid [not null]

  status varchar(20) [not null, default: 'PENDING', note: '''
    PENDING
    APPROVED
    REJECTED
  ''']

  comment text

  decided_at timestamptz

  created_at timestamptz [not null, default: `now()`]
}


// =====================================================
// RESERVATION CHANGE REQUEST
// =====================================================

Table reservation_change_requests {
  id uuid [pk]

  reservation_id uuid [not null]
  requested_by uuid [not null]

  change_data jsonb [not null]

  reason text

  status varchar(20) [not null, default: 'PENDING', note: '''
    PENDING
    APPROVED
    REJECTED
    CANCELLED
  ''']

  reviewed_by uuid
  reviewed_at timestamptz

  created_at timestamptz [not null, default: `now()`]

  indexes {
    (reservation_id, status)
  }
}


// =====================================================
// RETURN
// =====================================================

Table returns {
  id uuid [pk]

  reservation_id uuid [not null]

  returned_by uuid [not null]

  status varchar(20) [not null, default: 'PENDING', note: '''
    PENDING
    APPROVED
    REJECTED
    COMPLETED
  ''']

  returned_at timestamptz

  created_at timestamptz [not null, default: `now()`]
  updated_at timestamptz [not null, default: `now()`]

  indexes {
    (reservation_id)
    (status)
  }
}


// =====================================================
// RETURN ITEMS
// =====================================================

Table return_items {
  id uuid [pk]

  return_id uuid [not null]
  reservation_item_id uuid [not null]

  quantity_returned int [not null, default: 0]
  quantity_broken int [not null, default: 0]
  quantity_lost int [not null, default: 0]

  description text

  created_at timestamptz [not null, default: `now()`]

  indexes {
    (return_id, reservation_item_id) [unique]
  }
}
